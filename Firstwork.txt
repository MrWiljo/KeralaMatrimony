	@Test
	public void links() {
		
		List<WebElement> Nolinks = driver3.findElements(By.tagName("a"));		//links store in a list
		System.out.println("Number of Links :: "+Nolinks.size());
		System.out.println();
		
		for(WebElement e:Nolinks) {
			String w = e.getAttribute("href");			//printing the links
			System.out.println("* "+w);
			System.out.println();
			
			String l = e.getText();				//printing the name of links
			System.out.println(l);
			System.out.println();
		}
	}

		}

			System.out.println("FAILED :: The title of the site is DIFFERENT.");
		}
		System.out.println();
	}
